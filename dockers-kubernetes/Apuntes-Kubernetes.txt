// Aplicar un configfile
kubectl apply -f client-pod.yaml
// si quisiera aplicar todos los archivos de una carpeta
// siendo k8s el nombre de la carpeta
kubectl apply -f k8s

// Obtener objetos de Kubernetes
// podes cambiar pods por service, job, etc segun el objeto que quieras
kubectl get pods

// describir objeto
kubectl describe pod client-deployment-678c966f45-67z5j

// eliminar un objeto
kubectl delete -f client-deployment-678c966f45-67z5j
kubectl delete deployment client-deployment

// crear un secret
kubectl create secret generic pgpassword --from-literal PGPASSWORD=postgres

// como habilitar Kubernetes Dashboard via port forward
kubectl -n kubernetes-dashboard port-forward svc/kubernetes-dashboard-kong-proxy 8443:443
// si te pide un bearer token asi se genera
// entras a https://127.0.0.1:8443/ para ver el dashboard
kubectl -n kubernetes-dashboard create token admin-user
